# Lan Mouse implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use
> superpowers:subagent-driven-development (recommended) or
> superpowers:executing-plans to implement this plan task-by-task. Steps use
> checkbox (`- [ ]`) syntax for tracking.

**Goal:** Lan Mouse across 5 hosts, arranged from a Noctalia drag and drop
panel, with Alt+U to switch.

**Architecture:** Nix writes Lan Mouse's `config.toml` and installs a sops
certificate. A second `taildrop-syncd` instance syncs one layout JSON across
the tailnet. A Noctalia plugin applies that layout to the local daemon with
`lan-mouse cli` and edits it from a panel. A Hyprland Lua function handles
Alt+U.

**Tech stack:** Nix and Home Manager, sops-nix, Lan Mouse 0.11.0, Noctalia
5.3.0 Luau plugins, Hyprland 0.56.2 Lua config, Lua 5.5.

**Spec:** `docs/superpowers/specs/2026-10-17-lan-mouse-design.md`

## Global constraints

- Hosts: `karsa`, `kruppe`, `samar`, `mappo`, `tehol`. Baruk is excluded.
- Lan Mouse comes from nixpkgs (`pkgs.lan-mouse`, 0.11.0), not a flake input.
- `capture_backend = "layer-shell"`, `port = 4242`,
  `release_bind = ["KeyLeftAlt", "KeyU"]`,
  `cert_path = "/run/secrets/lan-mouse-cert"`.
- Layout sync runs `taildrop-syncd` with `TAILDROP_SYNCD_PORT=8472`.
- Fallback order for Alt+U: right, bottom, left, top.
- Commit straight to `main` with Conventional Commits. Do not stage
  `home/programs/zen/lib.nix`; another session owns it.
- Code comments follow `~/.claude/standards/comment-rules.md`: at most 2
  lines, only for constraints and gotchas.
- Deploy with deploy-rs: `deploy .#<host>`. Karsa can use
  `sudo nixos-rebuild switch --flake .#karsa`.

## Review focus

- Layout not loaded yet (layoutd down or still starting): the service must not
  touch Lan Mouse clients. Removing them all would drop every connection.
  Task 6 guards `reconcile` and its test pins `plan` input handling.
- A host name in the document that is not in the `hosts` setting (typo, or
  Baruk): it must never become a Lan Mouse client. Task 5 tests
  `Layout.view` with an unknown host.
- The Lan Mouse daemon restarts and loses its runtime clients: the next tick
  must add them all back. Task 5 tests `plan` with an empty client list.
- The last-used peer goes offline or is unplaced: Alt+U must fall back through
  right, bottom, left, top. Task 4 tests `pickTarget` for this.
- Mixed monitor sizes and negative positions, such as Tehol's external
  monitor at `-2560x0`: Alt+U must use the outermost monitor on that side.
  Task 4 tests `edgePoint` with Tehol's layout.

`home/programs/hypr/hyprland/lan_mouse.lua`:

```lua
local M = {}

local function readFile(path)
  local file = io.open(path, "r")
  if not file then
    return nil
  end
  local text = file:read("a")
  file:close()
  return text
end

function M.pickTarget(lastText, targetsText)
  local targets = {}
  for side, peer in (targetsText or ""):gmatch("(%a+) ([%w%-]+)") do
    table.insert(targets, { side = side, peer = peer })
  end
  local last = lastText and lastText:match("^%s*([%w%-]+)")
  for _, target in ipairs(targets) do
    if target.peer == last then
      return target
    end
  end
  return targets[1]
end

local function outer(m, side)
  if side == "right" then
    return m.x + m.w
  elseif side == "left" then
    return -m.x
  elseif side == "bottom" then
    return m.y + m.h
  end
  return -m.y
end

-- Lan Mouse's layer-shell strip is 1 pixel wide, so the cursor must move onto
-- it from inside the monitor to register an enter.
function M.edgePoint(monitors, side)
  local best
  for _, m in ipairs(monitors) do
    if not best or outer(m, side) > outer(best, side) then
      best = m
    end
  end
  if not best then
    return nil
  end
  local midX = best.x + best.w // 2
  local midY = best.y + best.h // 2
  if side == "right" then
    return { x = best.x + best.w - 3, y = midY }, { x = best.x + best.w - 1, y = midY }
  elseif side == "left" then
    return { x = best.x + 2, y = midY }, { x = best.x, y = midY }
  elseif side == "bottom" then
    return { x = midX, y = best.y + best.h - 3 }, { x = midX, y = best.y + best.h - 1 }
  end
  return { x = midX, y = best.y + 2 }, { x = midX, y = best.y }
end

function M.logical(monitor)
  local scale = monitor.scale > 0 and monitor.scale or 1
  local w = math.floor(monitor.size.width / scale + 0.5)
  local h = math.floor(monitor.size.height / scale + 0.5)
  if monitor.transform % 2 == 1 then
    w, h = h, w
  end
  return { x = monitor.position.x, y = monitor.position.y, w = w, h = h }
end

function M.push()
  local dir = (os.getenv("XDG_RUNTIME_DIR") or "/tmp") .. "/lan-mouse"
  local target = M.pickTarget(readFile(dir .. "/last"), readFile(dir .. "/targets"))
  if not target then
    hl.notification.create({ text = "Lan Mouse: no peer to switch to", timeout = 3000 })
    return
  end
  local monitors = {}
  for _, monitor in ipairs(hl.get_monitors()) do
    table.insert(monitors, M.logical(monitor))
  end
  local approach, edge = M.edgePoint(monitors, target.side)
  if approach then
    hl.dispatch(hl.dsp.cursor.move(approach))
    hl.dispatch(hl.dsp.cursor.move(edge))
  end
end

return M
```

```lua
local Layout = require("./layout")

local KNOWN = { karsa = true, tehol = true, mappo = true, kruppe = true, samar = true }

local function count(t)
  local n = 0
  for _ in pairs(t) do
    n += 1
  end
  return n
end

-- normalise drops bad links, self links and duplicate pairs.
local doc = Layout.normalise({
  links = {
    { host = "karsa", peer = "tehol", side = "right" },
    { host = "tehol", peer = "karsa", side = "left" },
    { host = "karsa", peer = "karsa", side = "left" },
    { host = "karsa", peer = "mappo", side = "diagonal" },
    "junk",
  },
})
assert(#doc.links == 1)
assert(count(doc.enabled) == 0 and count(doc.monitors) == 0)
assert(#Layout.normalise(nil).links == 0)

-- view works from both ends of a link.
assert(Layout.view(doc, "karsa", KNOWN).tehol == "right")
assert(Layout.view(doc, "tehol", KNOWN).karsa == "left")
assert(Layout.view(doc, "mappo", KNOWN).karsa == nil)

-- a disabled peer or a disabled self hides the link.
assert(Layout.view(Layout.setEnabled(doc, "tehol", false), "karsa", KNOWN).tehol == nil)
assert(count(Layout.view(Layout.setEnabled(doc, "karsa", false), "karsa", KNOWN)) == 0)
-- setEnabled copies; the original is unchanged.
assert(Layout.isEnabled(doc, "tehol"))

-- unknown hosts never appear.
local typo = Layout.normalise({ links = { { host = "karsa", peer = "baruk", side = "top" } } })
assert(Layout.view(typo, "karsa", KNOWN).baruk == nil)

-- place: onto an empty side, then onto an occupied side swaps.
```

```lua
--!nonstrict
local Layout = {}

local OPPOSITE = { left = "right", right = "left", top = "bottom", bottom = "top" }

Layout.FALLBACK_ORDER = { "right", "bottom", "left", "top" }

local function pairKey(a, b)
  return a < b and `{a}|{b}` or `{b}|{a}`
end

local function copyMap(t)
  local out = {}
  for k, v in pairs(type(t) == "table" and t or {}) do
    out[k] = v
  end
  return out
end

function Layout.normalise(doc)
  doc = type(doc) == "table" and doc or {}
  local links, seen = {}, {}
  for _, link in ipairs(type(doc.links) == "table" and doc.links or {}) do
    if
      type(link) == "table"
      and type(link.host) == "string"
      and type(link.peer) == "string"
      and link.host ~= link.peer
      and OPPOSITE[link.side]
      and not seen[pairKey(link.host, link.peer)]
    then
      seen[pairKey(link.host, link.peer)] = true
      table.insert(links, { host = link.host, peer = link.peer, side = link.side })
    end
  end
  return { enabled = copyMap(doc.enabled), links = links, monitors = copyMap(doc.monitors) }
end

function Layout.isEnabled(doc, host)
  return doc.enabled[host] ~= false
end

function Layout.sideOf(doc, me, peer)
  for _, link in ipairs(doc.links) do
    if link.host == me and link.peer == peer then
      return link.side
    elseif link.host == peer and link.peer == me then
      return OPPOSITE[link.side]
    end
  end
  return nil
end

local function otherEnd(link, me)
  if link.host == me then
    return link.peer
  elseif link.peer == me then
    return link.host
  end
  return nil
end

function Layout.view(doc, me, known)
  local view = {}
  if not Layout.isEnabled(doc, me) then
    return view
  end
  for _, link in ipairs(doc.links) do
    local peer = otherEnd(link, me)
    if peer and known[peer] and Layout.isEnabled(doc, peer) then
      view[peer] = Layout.sideOf(doc, me, peer)
    end
  end
  return view
end

local function withoutPair(links, a, b)
  local out = {}
  for _, link in ipairs(links) do
    if pairKey(link.host, link.peer) ~= pairKey(a, b) then
      table.insert(out, link)
    end
  end
  return out
end

function Layout.place(doc, me, peer, side)
  local next = Layout.normalise(doc)
  local previous = Layout.sideOf(next, me, peer)
  local occupant = nil
  if side then
    for _, link in ipairs(next.links) do
      local other = otherEnd(link, me)
      if other and other ~= peer and Layout.sideOf(next, me, other) == side then
        occupant = other
      end
    end
  end
  next.links = withoutPair(next.links, me, peer)
  if occupant then
    next.links = withoutPair(next.links, me, occupant)
    if previous then
      table.insert(next.links, { host = me, peer = occupant, side = previous })
    end
  end
  if side then
    table.insert(next.links, { host = me, peer = peer, side = side })
  end
  return next
end

function Layout.setEnabled(doc, host, on)
  local next = Layout.normalise(doc)
  next.enabled[host] = on
  return next
end

function Layout.setMonitors(doc, host, monitors)
  local next = Layout.normalise(doc)
  next.monitors[host] = monitors
  return next
end

function Layout.sameMonitors(a, b)
  if type(a) ~= "table" or type(b) ~= "table" or #a ~= #b then
    return false
  end
  for i, m in ipairs(a) do
    local n = b[i]
    if m.name ~= n.name or m.x ~= n.x or m.y ~= n.y or m.w ~= n.w or m.h ~= n.h then
      return false
    end
  end
  return true
end

function Layout.logicalMonitors(raw)
  local out = {}
  for _, m in ipairs(type(raw) == "table" and raw or {}) do
    if not m.disabled then
      local scale = (m.scale or 0) > 0 and m.scale or 1
      local w = math.floor(m.width / scale + 0.5)
      local h = math.floor(m.height / scale + 0.5)
      if (m.transform or 0) % 2 == 1 then
        w, h = h, w
      end
      table.insert(out, { name = m.name, x = m.x, y = m.y, w = w, h = h })
    end
  end
  table.sort(out, function(a, b)
    return a.name < b.name
  end)
  return out
end

function Layout.parseList(stdout)
  local clients = {}
  for line in stdout:gmatch("[^\n]+") do
    local id, host, pos, active = line:match("^id (%d+): (.-):%d+ %((%a+)%) active: (%a+)")
    if id then
      table.insert(clients, { id = tonumber(id), host = host, pos = pos, active = active == "true" })
    end
  end
  return clients
end

function Layout.plan(view, clients, addresses, hookFor)
  local commands, seen = {}, {}
  for _, client in ipairs(clients) do
    local side = view[client.host]
    if not side or seen[client.host] then
      table.insert(commands, { "remove-client", tostring(client.id) })
    else
      seen[client.host] = true
      if client.pos ~= side then
        table.insert(commands, { "set-position", tostring(client.id), side })
      end
      if not client.active then
        table.insert(commands, { "activate", tostring(client.id) })
      end
    end
  end
  local missing = {}
  for peer in pairs(view) do
    if not seen[peer] then
      table.insert(missing, peer)
    end
  end
  table.sort(missing)
  for _, peer in ipairs(missing) do
    local command = { "add-client", "--hostname", peer, "--enter-hook", hookFor(peer) }
    if addresses[peer] then
      table.insert(command, "--ips")
      table.insert(command, addresses[peer])
    end
    table.insert(commands, command)
  end
  return commands
end

function Layout.targets(view, online)
  local lines = {}
  for _, side in ipairs(Layout.FALLBACK_ORDER) do
    for peer, peerSide in pairs(view) do
      if peerSide == side and online[peer] then
        table.insert(lines, `{side} {peer}`)
      end
    end
  end
  return lines
end

return Layout
```

```lua
--!nonstrict
local Layout = require("./layout")

local TICK_MS = 5000

local doc = nil
local me = nil
local online = {}
local addresses = {}
local connected = false
local cliOk = true
local stream = nil
local busy = false
local lastRequest = 0

local function base()
  return `http://127.0.0.1:{noctalia.getConfig("layoutd_port")}`
end

local function runtimeDir()
  return `{noctalia.getenv("XDG_RUNTIME_DIR") or "/tmp"}/lan-mouse`
end

local function known()
  local set = {}
  for _, host in ipairs(noctalia.getConfig("hosts") or {}) do
    set[host] = true
  end
  return set
end

local function publish()
  noctalia.state.set("lanmouse.doc", doc)
  noctalia.state.set("lanmouse.me", me)
  noctalia.state.set("lanmouse.online", online)
  noctalia.state.set("lanmouse.connected", connected)
  noctalia.state.set("lanmouse.cli_ok", cliOk)
end

-- layoutd stores the layout JSON as the text of a taildrop-syncd revision.
local function decodeRevision(revision)
  local text = type(revision) == "table" and revision.text or ""
  return Layout.normalise(text ~= "" and noctalia.json.decode(text) or {})
end

local function write(next)
  local body = noctalia.json.encode({ text = noctalia.json.encode(next) })
  if not body then
    return
  end
  doc = next
  publish()
  noctalia.http({ url = `{base()}/doc`, method = "POST", body = body }, function() end)
end

local function hookFor(peer)
  local dir = runtimeDir()
  return `mkdir -p '{dir}' && printf %s {peer} > '{dir}/last'`
end

local function runAll(commands, done)
  local i = 0
  local function step()
    i += 1
    local command = commands[i]
    if not command then
      done()
      return
    end
    local argv = { noctalia.getConfig("lan_mouse_bin"), "cli" }
    for _, part in ipairs(command) do
      table.insert(argv, part)
    end
    noctalia.runAsync(argv, function(result)
      if result.exitCode ~= 0 then
        cliOk = false
      end
      step()
    end, 5000)
  end
  step()
end

local reconcile

-- A new client has no id until the next list, so one immediate rerun gives it
-- its side; `again` stops a failing add from looping.
reconcile = function(again)
  -- Before layoutd answers every client looks stale and would be removed.
  if busy or not doc or not me then
    return
  end
  busy = true
  local view = Layout.view(doc, me, known())
  noctalia.mkdirAll(runtimeDir())
  noctalia.writeFile(`{runtimeDir()}/targets`, table.concat(Layout.targets(view, online), "\n"))
  noctalia.runAsync({ noctalia.getConfig("lan_mouse_bin"), "cli", "list" }, function(result)
    if result.exitCode ~= 0 then
      cliOk, busy = false, false
      publish()
      return
    end
    cliOk = true
    local commands = Layout.plan(view, Layout.parseList(result.stdout or ""), addresses, hookFor)
    runAll(commands, function()
      busy = false
      publish()
      for _, command in ipairs(commands) do
        if command[1] == "add-client" and not again then
          reconcile(true)
          return
        end
      end
    end)
  end, 5000)
end

local function refreshHealth()
  noctalia.http({ url = `{base()}/health` }, function(response)
    if not (response.ok and response.status == 200) then
      return
    end
    local health = noctalia.json.decode(response.body)
    if not health then
      return
    end
    me = health.origin
    online, addresses = {}, {}
    for _, peer in ipairs(health.peers or {}) do
      online[peer.host] = peer.online == true
      addresses[peer.host] = peer.address
    end
    if me then
      online[me] = true
    end
    publish()
  end)
end

local function publishMonitors()
  if not doc or not me then
    return
  end
  noctalia.runAsync({ "hyprctl", "monitors", "-j" }, function(result)
    if result.exitCode ~= 0 then
      return
    end
    local monitors = Layout.logicalMonitors(noctalia.json.decode(result.stdout or "") or {})
    if doc and not Layout.sameMonitors(doc.monitors[me], monitors) then
      write(Layout.setMonitors(doc, me, monitors))
    end
  end, 5000)
end

local function connect()
  if stream then
    stream.stop()
    stream = nil
  end
  stream = noctalia.httpStream({ url = `{base()}/events` }, function(line)
    local payload = line:match("^data: (.*)$")
    if not payload then
      return
    end
    local revision = noctalia.json.decode(payload)
    if revision then
      doc, connected = decodeRevision(revision), true
      publish()
      reconcile(false)
    end
  end, function()
    connected, stream = false, nil
    publish()
  end)
  if not stream then
    connected = false
    publish()
  end
end

noctalia.state.watch("lanmouse.request", function(request)
  if type(request) ~= "table" or type(request.nonce) ~= "number" or request.nonce <= lastRequest then
    return
  end
  lastRequest = request.nonce
  if not doc or not me then
    return
  end
  if request.kind == "place" and type(request.peer) == "string" then
    write(Layout.place(doc, me, request.peer, request.side))
  elseif request.kind == "enable" and type(request.host) == "string" then
    write(Layout.setEnabled(doc, request.host, request.on == true))
  end
  reconcile(false)
end)

function update()
  if not stream then
    connect()
  end
  refreshHealth()
  publishMonitors()
  reconcile(false)
end

function onConfigChanged()
  connect()
  refreshHealth()
end

function onExit()
  if stream then
    stream.stop()
  end
end

noctalia.setUpdateInterval(TICK_MS)
publish()
connect()
refreshHealth()
```

```lua
  elseif not cliOk then

      placed += 1

    { key = noctalia.tr("tooltip.status"), value = status },
    { key = noctalia.tr("tooltip.peers"), value = tostring(placed) },
  })
  local color = (not connected or not cliOk) and "error" or (on and "on_surface" or "on_surface_variant")
  barWidget.render(ui.glyph({ name = on and "mouse" or "mouse-off", color = color }))
```

### Task 8: Drag and drop panel

**Files:**
- Modify: `home/programs/noctalia/plugins/lan-mouse/panel.luau`

**Interfaces:**
- Consumes: the `lanmouse.*` state (Task 6) and `Layout.sideOf`,
  `Layout.isEnabled` and `Layout.normalise` (Task 5).
- Produces: `lanmouse.request` writes: `{nonce, kind = "place", peer, side}`
  and `{nonce, kind = "enable", host, on}`.

- [ ] **Step 1: Write `panel.luau`.**

```lua
--!nonstrict
local Layout = require("./layout")

local DRAG_TYPE = "lan-mouse-peer"
local SLOT_W, SLOT_H = 140, 90
local CENTRE_W, CENTRE_H = 200, 110

local doc = noctalia.state.get("lanmouse.doc")
local me = noctalia.state.get("lanmouse.me")
local online = noctalia.state.get("lanmouse.online") or {}
local connected = noctalia.state.get("lanmouse.connected") == true

local function tr(key, subst)
  return noctalia.tr(key, subst)
end

local function request(payload)
  payload.nonce = noctalia.nowMs()
  noctalia.state.set("lanmouse.request", payload)
end

local function peers()
  local out = {}
  for _, host in ipairs(noctalia.getConfig("hosts") or {}) do
    if host ~= me then
      table.insert(out, host)
    end
  end
  return out
end

-- Draws monitors side by side to a shared scale; vertical offsets are dropped.
local function monitorStrip(host, maxW, maxH)
  local monitors = doc and doc.monitors[host] or {}
  local totalW, tallest = 0, 0
  for _, m in ipairs(monitors) do
    totalW += m.w
    tallest = math.max(tallest, m.h)
  end
  if totalW == 0 then
    return ui.box({ width = maxW, height = 4 })
  end
  local scale = math.min(maxW / totalW, maxH / tallest)
  local sorted = table.clone(monitors)
  table.sort(sorted, function(a, b)
    return a.x < b.x
  end)
  local boxes = {}
  for _, m in ipairs(sorted) do
    table.insert(boxes, ui.box({
      width = math.max(4, math.floor(m.w * scale)),
      height = math.max(4, math.floor(m.h * scale)),
      fill = "primary/0.25",
      radius = 3,
    }))
  end
  return ui.row({ gap = 2, align = "end" }, boxes)
end

local function tile(peer)
  local enabled = Layout.isEnabled(doc, peer)
  local isOnline = online[peer] == true
  local body = ui.column({ gap = 4, padding = 6, align = "stretch" }, {
    ui.row({ gap = 4, align = "center" }, {
      ui.glyph({ name = "circle-filled", size = 8, color = isOnline and "primary" or "outline" }),
      ui.label({
        text = peer,
        fontSize = 12,
        color = enabled and "on_surface" or "on_surface_variant",
        flexGrow = 1,
      }),
      ui.button({
        glyph = "power",
        variant = "ghost",
        tooltip = enabled and tr("panel.disable") or tr("panel.enable"),
        onClick = function()
          request({ kind = "enable", host = peer, on = not enabled })
        end,
      }),
    }),
    monitorStrip(peer, SLOT_W - 16, 36),
  })
  if not enabled then
    return ui.box({ key = `tile-{peer}`, width = SLOT_W, fill = "on_surface/0.04", radius = 8 }, { body })
  end
  return ui.dragSource({
    key = `tile-{peer}`,
    dragType = DRAG_TYPE,
    payload = peer,
    previewAncestor = 0,
    liftFromLayout = true,
    width = SLOT_W,
  }, { ui.box({ fill = "on_surface/0.08", radius = 8 }, { body }) })
end

local function occupant(side)
  for _, peer in ipairs(peers()) do
    if Layout.isEnabled(doc, peer) and Layout.sideOf(doc, me, peer) == side then
      return peer
    end
  end
  return nil
end

local function slot(side)
  local peer = occupant(side)
  local child = peer and tile(peer)
    or ui.label({ text = tr("panel.drop"), fontSize = 10, color = "on_surface_variant" })
  return ui.dropZone({
    key = `slot-{side}`,
    accepts = { DRAG_TYPE },
    value = side,
    onDrop = "onPeerDrop",
    width = SLOT_W,
    height = SLOT_H,
    radius = 8,
  }, { child })
end

local function centre()
  return ui.column({ width = CENTRE_W, height = CENTRE_H, gap = 4, align = "center", justify = "center" }, {
    monitorStrip(me, CENTRE_W - 20, CENTRE_H - 30),
    ui.label({ text = tr("panel.this_host", { host = me }), fontSize = 11 }),
  })
end

local function unplaced()
  local tiles = {}
  for _, peer in ipairs(peers()) do
    if not Layout.isEnabled(doc, peer) or not Layout.sideOf(doc, me, peer) then
      table.insert(tiles, tile(peer))
    end
  end
  return ui.dropZone({
    key = "slot-none",
    accepts = { DRAG_TYPE },
    value = "none",
    onDrop = "onPeerDrop",
    minHeight = SLOT_H,
    radius = 8,
  }, {
    ui.column({ gap = 4, padding = 6, align = "stretch" }, {
      ui.label({ text = tr("panel.unplaced"), fontSize = 11, color = "on_surface_variant" }),
      ui.row({ gap = 6 }, tiles),
    }),
  })
end

local function header(selfOn)
  local status = not connected and tr("status.offline") or (selfOn and tr("status.ok") or tr("status.off"))
  return ui.row({ gap = 6, align = "center" }, {
    ui.glyph({ name = "mouse", color = connected and "primary" or "error" }),
    ui.label({ text = tr("panel.title"), fontSize = 13, fontWeight = "semibold" }),
    ui.label({ text = status, fontSize = 11, color = "on_surface_variant" }),
    ui.spacer({ flexGrow = 1 }),
    ui.toggle({ checked = selfOn, onChange = "onToggleSelf" }),
  })
end

local function render()
  if not doc or not me or not connected then
    panel.render(ui.column({ flexGrow = 1, padding = 12, gap = 8 }, {
      header(false),
      ui.label({ text = tr("status.offline"), color = "on_surface_variant" }),
    }))
    return
  end
  local selfOn = Layout.isEnabled(doc, me)
  local grid = ui.column({ gap = 8, align = "center" }, {
    ui.row({ gap = 8 }, { ui.box({ width = SLOT_W }), slot("top"), ui.box({ width = SLOT_W }) }),
    ui.row({ gap = 8, align = "center" }, { slot("left"), centre(), slot("right") }),
    ui.row({ gap = 8 }, { ui.box({ width = SLOT_W }), slot("bottom"), ui.box({ width = SLOT_W }) }),
  })
  panel.render(ui.column({ flexGrow = 1, padding = 12, gap = 10, align = "stretch" }, {
    header(selfOn),
    grid,
    unplaced(),
  }))
end

for _, key in ipairs({ "lanmouse.doc", "lanmouse.me", "lanmouse.online", "lanmouse.connected" }) do
  noctalia.state.watch(key, function(value)
    if key == "lanmouse.doc" then
      doc = value and Layout.normalise(value)
    elseif key == "lanmouse.me" then
      me = value
    elseif key == "lanmouse.online" then
      online = value or {}
    else
      connected = value == true
    end
    render()
  end)
end

function onPeerDrop(payload, value)
  request({ kind = "place", peer = payload, side = value ~= "none" and value or nil })
end

function onToggleSelf(value)
  request({ kind = "enable", host = me, on = value == true })
end

function onOpen()
  render()
end

if doc then
  doc = Layout.normalise(doc)
end
render()
```
